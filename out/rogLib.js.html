function rogFmtFecha(f = new Date()) {
  const dicc = {
    fmts: {
      dd: () => rogFmt.dd(elementos.dia),
      ddd: () => trad("dia", elementos.nbDia).slice(0, 3),
      dddd: () => trad("dia", elementos.nbDia),
      mmmm: () => trad("mes", elementos.nbMes),
      mmm: () => trad("mes", elementos.nbMes).slice(0, 3),
      mm: () => rogFmt.dd(indice(elementos.nbMes, dicc.mes) + 1),
      aa: () => elementos.ano.slice(2),
      aaaa: () => elementos.ano,
      hh: () => elementos.hora.slice(0, 5),
      hhhh: () => elementos.hora,
      amd: (separador = "-") =>
        fmt("aaaa") + separador + fmt("mm") + separador + fmt("dd"),
      am: () => intp("aaaa / mm").replace(/[\s]/g, ""),
      dma: (separador = "/") =>
        fmt("dd") + separador + fmt("mm") + separador + fmt("aaaa"),
      cool: () => intp("ddd dd de mmm aaaa"),
      coolfull: () => intp("ddd dd de mmm aaaa hh"),
      serial: () => intp("aa mm dd hhhh").replace(/[\s:]/g, ""),
      full: () => intp("aa mm dd hhhh"),
      serio: () => intp("dddd dd de mmmm de aaaa"),
      evento: () => intp("ddd dd hh"),
      nbMes: (corto = false) => intp(corto ? "mmm aa" : "mmmm de aaaa"),
      lstMeses: () => Object.entries(dicc.mes).map(x => x[1]),
      lstDias: () => Object.entries(dicc.dia).map(x => x[1])
    },
    mes: {
      Jan: "Enero",
      Feb: "Febrero",
      Mar: "Marzo",
      Apr: "Abril",
      May: "Mayo",
      Jun: "Junio",
      Jul: "Julio",
      Aug: "Agosto",
      Sep: "Septiembre",
      Oct: "Octubre",
      Nov: "Noviembre",
      Dic: "Diciembre"
    },
    dia: {
      Sun: "Domingo",
      Mon: "Lunes",
      Tue: "Martes",
      Wed: "Miércoles",
      Thu: "Jueves",
      Fri: "Viernes",
      Sat: "Sábado"
    }
  };

  const indice = (valor, obj) => Object.keys(obj).findIndex((x) => x === valor);
  const fmt = (que) => (dicc.fmts[que] ? dicc.fmts[que]() : que);
  const trad = (que, valor) => dicc[que][valor];
  const intp = (cadena) => cadena.split(" ").map(fmt).join(" ");

  const elementos = parseFecha(creaFecha(f));

  function parseFecha(f) {
    const cadenas = f.toString().split(" ");
    return ["nbDia", "nbMes", "dia", "ano", "hora"].reduce((el, x, i) => {
      el[x] = cadenas[i];
      return el;
    }, {});
  }

  return dicc.fmts;
}

function creaFecha(ano, mes, dia) {
        // Devuelve un objeto fecha a partir de los parámetros enviados
        // si no se envía nada, devuelve la fecha de hoy
        var fecha;
        
        if (mes === undefined) {
            if (ano === undefined) {
              let hoy = new Date();
              if(dia === undefined) fecha = hoy;
              else fecha = new Date(hoy.getFullYear(), hoy.getMonth(), dia);
            } else {
                if (typeof(ano) === "object") fecha = ano instanceof Date ? ano : ano.fecha()
                else {fecha = new Date(ano) }
            }
        } else {
            if (ano === undefined) fecha = new Date(ano,mes);
            else fecha = new Date(ano, mes, dia || 1);
        }
        return fecha.toString() === "Invalid Date" ? undefined : fecha;
}

function strCompara(campo,descendente){
    /*
        Retorna una función que compara dos objetos sobre la base de uno o varios de sus elementos
        @param campo      : es la lista de elementos a comparar
        @param descendente: si es verdadero, la comparación es descendente 

        Uso: para el método "sort" de un arreglo de objetos
    */
    
    var menor = -1, mayor = 1;
    var que = campo;
    const comp1 = (a,b) => (def(a[que]) < def(b[que])) ? menor : (def(a[que]) > def(b[que])) ? mayor : 0;
    let _campos = desarma(que);
    function comp(a,b) {
      let i = 0;
      do {
        que = _campos[i];
        retorno = comp1(a,b);
      } while (++i < _campos.length && !retorno);
      return retorno;
    }
    if (descendente) {
        menor = 1;
        mayor = -1;
    }
    if (_campos.length === 1) return comp1;
    else return comp;
}

function desarma(lista) {
    // Crea un arreglo a partir de:
    // 1. Una lista de palabras separada por coma y un espacio en una cadena de caracteres: "A, B, C"
    // 2. Un objeto: { A: ..., B: ..., C: ... }
    // 3. Si @param valores es Un arreglo devuelve la referencia correspondiente

    let retorno = [];
    if(lista) {
        if(lista instanceof Array) retorno = lista;
        else if(typeof lista === "string") retorno = lista.split(", ");
        else retorno = Object.keys(lista);
    }
    return retorno;
}

function accede(accion,url,fn,datos) {
    /**
        Ejecuta @param {string} acción http sobre la dirección @param {string} url
        Envía @param {object} datos como "body" de la acción

        Según el retorno:
        Si hay un error o el retorno tiene la propiedad "fallo" en "true"
        Se muestra un mensaje (alert)
        
        Se llama @param {function} fn con el retorno como argumento, dependiendo del error

        IDENTIFICACIÓN
        Se usa el localStorage para controlar la sesión de trabajo con JWT
    */
    let xobj = new XMLHttpRequest();
    let xmlEstado = ["No inicializado", "Conectado", "Recibido", "Procesando", "Listo"];

	function retorno(xobj) {
        if(xobj.tipoResultado === "error") {
            alert(msjError(xobj));
            rogFallo(objError(xobj));
        } else {
            if(xobj.tipoResultado === "página") neaVnt(xobj.resultado);
            else {
                if(fn) fn(xobj.resultado);
                else alert(xobj.resultado);
            }
        }
    }

    function analizaResultado(xobj) {
        xobj.tipoResultado = typeof xobj.responseText === "string" ?
            xobj.responseText.startsWith("<!") ?
                "página" : "respuesta" :
                "objeto";

        xobj.resultado = xobj.tipoResultado === "respuesta" ?
            JSON.parse(xobj.responseText) :
            xobj.responseText;

        if (huboError(xobj)) xobj.tipoResultado = "error" 
    }

    function msjError(xobj) {
        return (xobj.resultado && xobj.resultado.fallo) ?
            (xobj.resultado.msj instanceof Array ? xobj.resultado.msj.join("\n") : xobj.resultado.msj) :
            `${xobj.accion}\n${xobj.url}\n¡E R R O R!!!\nStatus: ${xobj.status} (${xobj.statusText})\n${xobj.status}`;
    }

    function huboError(xobj) {
        return ((xobj.status == 0) || (xobj.status >= 400)) || (xobj.resultado && xobj.resultado.fallo);
    }

	function objError(xobj) {
        return (xobj.resultado && xobj.resultado.fallo) ?
		    xobj.resultado :
            { 
                fallo: true, 
                msj: xobj.statusText || "No puedo comunicarme con la B.D.", 
                status: (xobj.resultado && xobj.resultado.status) || xobj.status
            }
	}

    function mstError(e) {
        retorno({tipoResultado: "error", ...e.target});
    }

    function hayQueIdentificar(xobj) {
        return [400,401,491,492].includes(xobj.status);        
    }

    xobj.accion = accion;
    xobj.url    = url;

    xobj.overrideMimeType("application/json");
    xobj.withCredentials = true;
    xobj.open(accion, url, true);
    xobj.setRequestHeader('Access-Control-Allow-Origin',inception);
    xobj.setRequestHeader('Content-type','application/json; charset=utf-8');
    /*debugger*/
    let token = localStorage.getItem(API);
    if(token) xobj.setRequestHeader("autoriza",token);
    xobj.onerror = mstError;
    xobj.onreadystatechange = () => console.log(`${xobj.readyState} ${url}, ${xmlEstado[xobj.readyState]} (${xobj.status})`);
    xobj.onload = () => {
        analizaResultado(xobj);

        identifica(xobj);

        retorno(xobj);
	};
    xobj.send(JSON.stringify(datos));
}

function creaCmb(datos,dom,item) {
    // Crea el dom de un combo cargado con @param datos en el @param dom
    // Si se omite @param dom se devuelve el codigo HTML correspondiente
    // @param item puede ser:
    // 1. el nombre de un elemento en datos
    // 2. un Objeto { valor: texto } o {valor: nb del valor, texto: nb del texto} con el nombre de los elementos a usar
    // 3. Si se omite, es el valor del elemento en datos

    let valor = (x,i) => i;
    let texto = x => x;
    if(item) {
        if(typeof item === "string") {
            valor = x => x[item];
            texto = x => x[item];
        } else {
            valor = (x,i) => x[item.valor] || i;
            texto = x => x[item.texto]; 
        }
    }
    
    let _combo = datos.reduce((combo,x,i)=> combo +="<option value='"+valor(x,i)+"'>"+texto(x)+"</option>","<select>")+"</select>";
    if(dom) {
        let _dom = objDom(dom);
        _dom.innerHTML = _combo;
    } else return _combo;
}

        else {
            mstMsj(msj.msj);

                //campoError(msj.campo).setCustomValidity( msj.msj);
                campoMsjError(msj.campo).innerHTML = msj.msj;

        }

    if(msj.msjs) {
